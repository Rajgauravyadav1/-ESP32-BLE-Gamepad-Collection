#include <BleGamepad.h>

#define BTN_A   12
#define BTN_B   13
#define BTN_X   14
#define BTN_Y   27
#define JOY_X   34
#define JOY_Y   35

#define DEADZONE  200
#define CENTER    2048

BleGamepad bleGamepad("ESP32 4B+J", "DIY", 100);
BleGamepadConfiguration cfg;

bool prev[4] = {false};
const int pins[4] = {BTN_A,BTN_B,BTN_X,BTN_Y};

int16_t mapJoy(int32_t raw) {
  int32_t c = raw - CENTER;
  if (abs(c) < DEADZONE) return 0;
  return (c > 0)
    ? map(c,DEADZONE,CENTER,0,32767)
    : map(c,-CENTER,-DEADZONE,-32767,0);
}

void setup() {
  for(int i=0;i<4;i++)
    pinMode(pins[i], INPUT_PULLUP);
  cfg.setAutoReport(false);
  cfg.setButtonCount(4);
  cfg.setIncludeXAxis(true);
  cfg.setIncludeYAxis(true);
  cfg.setAxesMin(-32767);
  cfg.setAxesMax(32767);
  bleGamepad.begin(&cfg);
}

void loop() {
  if(!bleGamepad.isConnected()){delay(100);return;}
  for(int i=0;i<4;i++){
    bool cur=!digitalRead(pins[i]);
    if(cur!=prev[i]){
      cur?bleGamepad.press(i+1)
         :bleGamepad.release(i+1);
      prev[i]=cur;
    }
  }
  bleGamepad.setX(mapJoy(analogRead(JOY_X)));
  bleGamepad.setY(mapJoy(analogRead(JOY_Y)));
  bleGamepad.sendReport();
  delay(10);
}

Arduino code
#include <BleGamepad.h>

// 8 buttons + joystick + 2 trigger switches
const int BTN_PINS[] = {12,13,14,27,26,25,33,32};
const int BTN_COUNT  = 8;
#define JOY_X    34
#define JOY_Y    35
#define LT_PIN   36  // trigger switch
#define RT_PIN   39  // trigger switch
#define DEADZONE 200
#define CENTER   2048

BleGamepad bleGamepad("ESP32 8B+J+TRG","DIY",100);
BleGamepadConfiguration cfg;
bool prev[8]={false};

int16_t mapJoy(int32_t raw){
  int32_t c=raw-CENTER;
  if(abs(c)<DEADZONE) return 0;
  return(c>0)?map(c,DEADZONE,CENTER,0,32767)
              :map(c,-CENTER,-DEADZONE,-32767,0);
}

int16_t mapTrig(int32_t raw){
  return (int16_t)map(
    constrain(raw,0,4095),0,4095,0,32767);
}

void setup(){
  for(int i=0;i<BTN_COUNT;i++)
    pinMode(BTN_PINS[i],INPUT_PULLUP);
  cfg.setAutoReport(false);
  cfg.setButtonCount(BTN_COUNT);
  cfg.setIncludeXAxis(true);
  cfg.setIncludeYAxis(true);
  cfg.setIncludeZAxis(true);   // LT
  cfg.setIncludeRzAxis(true);  // RT
  cfg.setAxesMin(-32767);
  cfg.setAxesMax(32767);
  bleGamepad.begin(&cfg);
}

void loop(){
  if(!bleGamepad.isConnected()){delay(100);return;}
  for(int i=0;i<BTN_COUNT;i++){
    bool cur=!digitalRead(BTN_PINS[i]);
    if(cur!=prev[i]){
      cur?bleGamepad.press(i+1)
         :bleGamepad.release(i+1);
      prev[i]=cur;
    }
  }
  // Joystick
  bleGamepad.setX(mapJoy(analogRead(JOY_X)));
  bleGamepad.setY(mapJoy(analogRead(JOY_Y)));
  // Triggers → Z / RZ axes
  bleGamepad.setZ(mapTrig(analogRead(LT_PIN)));
  bleGamepad.setRZ(mapTrig(analogRead(RT_PIN)));
  bleGamepad.sendReport();
  delay(10);
}

// Vibration Module (S8050 NPN)
// IN pin driven directly by ESP32 GPIO
// Use ledcWrite() for PWM intensity control

#define MOTOR1_PIN  25
#define MOTOR2_PIN  26

#define PWM_FREQ    5000   // Hz
#define PWM_RES     8      // bits (0-255)
#define CH_M1       0
#define CH_M2       1

void setup() {
  ledcSetup(CH_M1, PWM_FREQ, PWM_RES);
  ledcSetup(CH_M2, PWM_FREQ, PWM_RES);
  ledcAttachPin(MOTOR1_PIN, CH_M1);
  ledcAttachPin(MOTOR2_PIN, CH_M2);
}

// intensity: 0 = off, 255 = full speed
void setMotor1(uint8_t intensity) {
  ledcWrite(CH_M1, intensity);
}
void setMotor2(uint8_t intensity) {
  ledcWrite(CH_M2, intensity);
}

void rumble(uint8_t left, uint8_t right) {
  setMotor1(left);
  setMotor2(right);
}

void loop() {
  rumble(200, 200);  // both on ~78%
  delay(500);
  rumble(0, 0);      // both off
  delay(1000);
}

// TB6612FNG — Dual H-bridge
// Supports speed + direction + brake

// Motor 1 (Channel A)
#define PWMA   25
#define AIN1   26
#define AIN2   27
// Motor 2 (Channel B)
#define PWMB   32
#define BIN1   33
#define BIN2   14
// Enable
#define STBY   -1  // wired to 3.3V directly

#define PWM_FREQ  10000
#define PWM_RES   8
#define CH_A      0
#define CH_B      1

void setup() {
  pinMode(AIN1, OUTPUT); pinMode(AIN2, OUTPUT);
  pinMode(BIN1, OUTPUT); pinMode(BIN2, OUTPUT);
  ledcSetup(CH_A, PWM_FREQ, PWM_RES);
  ledcSetup(CH_B, PWM_FREQ, PWM_RES);
  ledcAttachPin(PWMA, CH_A);
  ledcAttachPin(PWMB, CH_B);
}

// speed 0–255, forward only for vibration
void motor1(uint8_t speed) {
  digitalWrite(AIN1, HIGH);
  digitalWrite(AIN2, LOW);
  ledcWrite(CH_A, speed);
}
void motor2(uint8_t speed) {
  digitalWrite(BIN1, HIGH);
  digitalWrite(BIN2, LOW);
  ledcWrite(CH_B, speed);
}
// Brake: both IN pins HIGH
void brakeAll() {
  digitalWrite(AIN1, HIGH); digitalWrite(AIN2, HIGH);
  digitalWrite(BIN1, HIGH); digitalWrite(BIN2, HIGH);
  ledcWrite(CH_A, 0); ledcWrite(CH_B, 0);
}

void rumble(uint8_t left, uint8_t right) {
  motor1(left);
  motor2(right);
}

void loop() {
  rumble(200, 200);
  delay(500);
  brakeAll();
  delay(1000);
}

// DRV8833 — Texas Instruments
// 3.3V only, nFAULT fault detection
// AIN1=PWM, AIN2=LOW for forward spin

#define AIN1      25  // Motor 1 PWM
#define AIN2      26  // Motor 1 dir
#define BIN1      27  // Motor 2 PWM
#define BIN2      14  // Motor 2 dir
#define NFAULT    35  // LOW = fault
// nSLEEP wired to 3.3V directly

#define PWM_FREQ  20000  // inaudible
#define PWM_RES   8
#define CH_A      0
#define CH_B      1

void setup() {
  Serial.begin(115200);
  pinMode(AIN2, OUTPUT); digitalWrite(AIN2, LOW);
  pinMode(BIN2, OUTPUT); digitalWrite(BIN2, LOW);
  pinMode(NFAULT, INPUT_PULLUP);
  ledcSetup(CH_A, PWM_FREQ, PWM_RES);
  ledcSetup(CH_B, PWM_FREQ, PWM_RES);
  ledcAttachPin(AIN1, CH_A);
  ledcAttachPin(BIN1, CH_B);
}

bool isFault() {
  return !digitalRead(NFAULT);
}

void motor1(uint8_t speed) {
  ledcWrite(CH_A, speed);
}
void motor2(uint8_t speed) {
  ledcWrite(CH_B, speed);
}

void rumble(uint8_t left, uint8_t right) {
  if (isFault()) {
    Serial.println("DRV8833 fault!");
    return;
  }
  motor1(left);
  motor2(right);
}

void loop() {
  rumble(200, 200);
  delay(500);
  motor1(0); motor2(0);
  delay(1000);
}
